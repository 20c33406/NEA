class Vector 
    public X
    public Y
    public procedure new(v_x,v_y)
        X=v_x

    endprocedure
    procedure addVector(vector)
        X += vector.X
        Y += vector.Y
    endprocedure

    procedure setZero()
        X = 0
        Y = 0
    endprocedure
endclass

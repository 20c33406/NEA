class Vector 
    public X
    public Y
    public procedure new(v_x,v_y)
        this.X=v_x
        this.Y=v_y
    endprocedure
    procedure addVector(vector)
        this.X += vector.X
        this.Y += vector.Y
    endprocedure

    procedure setZero()
        this.X = 0
        this.Y = 0
    endprocedure
endclass
